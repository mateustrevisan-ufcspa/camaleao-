# US-31 · Comissão por doação, com quem captou (8 SP)

> Como coordenadora do financeiro, eu quero registrar em cada doação quem captou e o valor da comissão, e somar o mês todo ou só algumas doações, para que o resultado do mês no sistema bata com a planilha do Instituto.

**Tipo:** user story · **Implementa:** Bibiana · **Revisa:** Mateus · **Branch:** `us-31-comissao-por-doacao` · **Depende de:** US-35 integrada (a lista mostra o líquido) e SEC-04 (a comissão entra na trilha) · **PR pronto:** quinta 12/11, 18h · **Merge:** quinta 12/11, à noite, depois da US-21a

**Reestimada de 5 para 8 SP** no planejamento de 06/11. O cartão previa um lançamento por mês; a Nílian respondeu (03/11) que a comissão é **por doação**, para quem captou ("hoje à Ana, mas pode haver outras pessoas na captação"), e que querem "ver o valor por doação e somar o mês todo ou só algumas doações". Isso pede o cadastro de quem capta, o valor em cada doação, a soma com seleção, e o relatório e a exportação.

## Critérios de aceitação

- [ ] A coordenação cadastra quem capta (nome) e desativa quem parou; quem foi desativado continua nas comissões antigas
- [ ] Em cada doação em dinheiro do mês, a coordenação escolhe quem captou e informa o valor da comissão (preenchido à mão, sem cálculo pelo sistema), e corrige ou remove depois
- [ ] A tela soma a comissão do mês inteiro, por pessoa que captou, e das doações que a coordenação marcar
- [ ] O relatório consolidado e a exportação mostram as comissões do período e o resultado depois delas
- [ ] Quem lançou e quando vêm do banco, e cada correção entra na trilha de auditoria (SEC-04)
- [ ] O balcão não vê nem altera comissão nem quem capta; testes de acesso e de fumaça passam

## Decisões

- **Tabela própria, e não coluna na doação:** `comissoes_de_doacao`, uma linha por doação, com a política "só a coordenação". O caixa lê as doações do dia (US-28) e, com coluna na própria doação, leria a comissão também.
- **Quem capta é um cadastro simples** (`captadores`: nome e ativo), e não uma conta do sistema: a Nílian disse que pode haver outras pessoas na captação, e nem todas usam o sistema. Sem nome de ninguém na migração; a coordenação cadastra a Ana na tela.
- **Só doações em dinheiro.** Comissão de vendas não apareceu na resposta; está na pergunta 7 da mensagem de 06/11. Se vier, é a mesma tabela com `sale_id`, numa story nova.
- **Marcar para somar** é só na tela (não grava). A Nílian quer "somar o mês todo ou só algumas doações", para conferir com a planilha; pagamento da comissão não foi pedido.
- **Doação cancelada (US-04b) sai das somas sozinha:** a view das comissões junta com `donations_cash`, que esconde as canceladas pela política do banco.

```sql
-- US-31 · Comissão por doação, com quem captou
-- Cadastro de quem capta, a comissão de cada doação em dinheiro (preenchida pela
-- coordenação, sem cálculo) e a view para as somas. Só a coordenação.

-- ─────────────────────────────────────────
-- 1. Quem capta. Desativar em vez de apagar: as comissões antigas apontam para cá.
-- ─────────────────────────────────────────

create table public.captadores (
  id         uuid primary key default gen_random_uuid(),
  nome       text not null check (length(btrim(nome)) between 2 and 80),
  ativo      boolean not null default true,
  criado_em  timestamptz not null default now(),
  criado_por uuid default auth.uid() references public.users(id) on delete set null
);

create unique index idx_captadores_nome on public.captadores (lower(btrim(nome)));

alter table public.captadores enable row level security;

create policy "coordenacao le" on public.captadores
  for select to authenticated using (public.eh_coordenacao());
create policy "coordenacao cadastra" on public.captadores
  for insert to authenticated with check (public.eh_coordenacao());
create policy "coordenacao altera" on public.captadores
  for update to authenticated using (public.eh_coordenacao()) with check (public.eh_coordenacao());

revoke all on public.captadores from anon;
grant select, insert, update on public.captadores to authenticated;

-- ─────────────────────────────────────────
-- 2. A comissão de cada doação. Uma por doação; some com a doação apagada.
-- ─────────────────────────────────────────

create table public.comissoes_de_doacao (
  id           uuid primary key default gen_random_uuid(),
  donation_id  uuid not null unique references public.donations_cash(id) on delete cascade,
  captador_id  uuid not null references public.captadores(id) on delete restrict,
  valor        numeric(10,2) not null check (valor >= 0 and valor <= 100000),
  definido_em  timestamptz not null default now(),
  definido_por uuid references public.users(id) on delete set null
);

create index idx_comissoes_captador on public.comissoes_de_doacao (captador_id);

-- Quem e quando vêm do banco, também na correção.
create or replace function public.carimbar_comissao()
returns trigger
language plpgsql
set search_path = ''
as $$
begin
  new.definido_em := now();
  new.definido_por := auth.uid();
  return new;
end;
$$;

create trigger carimbo_da_comissao
  before insert or update on public.comissoes_de_doacao
  for each row execute function public.carimbar_comissao();

-- SEC-04: correção e remoção de comissão entram na trilha.
create trigger trilha_de_auditoria before update or delete on public.comissoes_de_doacao
  for each row execute function public.registrar_na_trilha();

alter table public.comissoes_de_doacao enable row level security;

create policy "coordenacao le" on public.comissoes_de_doacao
  for select to authenticated using (public.eh_coordenacao());
create policy "coordenacao lanca" on public.comissoes_de_doacao
  for insert to authenticated with check (public.eh_coordenacao());
create policy "coordenacao corrige" on public.comissoes_de_doacao
  for update to authenticated using (public.eh_coordenacao()) with check (public.eh_coordenacao());
create policy "coordenacao remove" on public.comissoes_de_doacao
  for delete to authenticated using (public.eh_coordenacao());

revoke all on public.comissoes_de_doacao from anon;
grant select, insert, update, delete on public.comissoes_de_doacao to authenticated;

-- ─────────────────────────────────────────
-- 3. Comissões com a data da doação e o nome de quem captou. security_invoker:
--    só a coordenação recebe linhas, e doação cancelada (US-04b) não aparece.
-- ─────────────────────────────────────────

create view public.comissoes_view
with (security_invoker = true) as
select
  k.id,
  k.donation_id,
  k.captador_id,
  c.nome as captador_nome,
  k.valor,
  k.definido_em,
  d.donated_at,
  d.donor_name,
  d.amount,
  d.net_amount
from public.comissoes_de_doacao k
join public.captadores c on c.id = k.captador_id
join public.donations_cash d on d.id = k.donation_id;

revoke all on public.comissoes_view from anon;
grant select on public.comissoes_view to authenticated;
```

```ts
// As doações em dinheiro do mês, com a comissão de cada uma (quando houver).
export async function getDoacoesComComissao(ano: number, mes: number): Promise<DoacaoComComissao[]> {
  const desde = brMidnightUtc(ano, mes, 1).toISOString()
  const ate = (mes === 12 ? brMidnightUtc(ano + 1, 1, 1) : brMidnightUtc(ano, mes + 1, 1)).toISOString()
  const supabase = await createClient()
  const [doacoes, comissoes] = await Promise.all([
    supabase.from('donations_cash_view').select('id, donated_at, donor_name, amount, net_amount').gte('donated_at', desde).lt('donated_at', ate).order('donated_at'),
    supabase.from('comissoes_view').select('donation_id, captador_id, captador_nome, valor').gte('donated_at', desde).lt('donated_at', ate),
  ])
  const erro = doacoes.error ?? comissoes.error
  if (erro) throw new Error(erro.message)
  const porDoacao = new Map((comissoes.data ?? []).map((k) => [k.donation_id as string, k]))
  return (doacoes.data ?? []).map((d) => {
    const k = porDoacao.get(d.id as string)
    return {
      id: d.id as string,
      donated_at: d.donated_at as string,
      donor_name: d.donor_name as string,
      amount: Number(d.amount),
      net_amount: d.net_amount === null ? null : Number(d.net_amount),
      captador_id: (k?.captador_id as string) ?? null,
      captador_nome: (k?.captador_nome as string) ?? null,
      comissao: k ? Number(k.valor) : null,
    }
  })
}
```

## Parte D · `actions/comissoes.ts` (arquivo novo)

```ts
'use server'

import { revalidatePath } from 'next/cache'
import { addCaptador, removeComissao, setCaptadorAtivo, setComissao, souCoordenacao } from '@/lib/store'
import { parseMoney } from '@/lib/utils'

const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i
const AVISO_RESTRITO = 'Só a coordenação lança comissões.'

export async function cadastrarCaptador(fd: FormData): Promise<{ error: string } | void> {
  if (!(await souCoordenacao())) return { error: AVISO_RESTRITO }
  const nome = ((fd.get('nome') as string) || '').replace(/\s+/g, ' ').trim()
  if (nome.length < 2 || nome.length > 80) return { error: 'Informe o nome de quem capta.' }
  try {
    await addCaptador(nome)
  } catch (e) {
    if ((e as { code?: string }).code === '23505') return { error: 'Já existe alguém com este nome.' }
    console.error('cadastrarCaptador error:', e)
    return { error: 'Não foi possível cadastrar.' }
  }
  revalidatePath('/relatorios/comissoes')
}

export async function alterarCaptador(id: string, ativo: boolean): Promise<{ error: string } | void> {
  if (!(await souCoordenacao())) return { error: AVISO_RESTRITO }
  if (!UUID.test(id)) return { error: 'Pessoa não encontrada.' }
  await setCaptadorAtivo(id, ativo)
  revalidatePath('/relatorios/comissoes')
}

export async function salvarComissao(donationId: string, fd: FormData): Promise<{ error: string } | void> {
  if (!(await souCoordenacao())) return { error: AVISO_RESTRITO }
  const captadorId = ((fd.get('captador_id') as string) || '').trim()
  const digitado = ((fd.get('valor') as string) || '').trim()
  // parseMoney devolve 0 para texto inválido: só aceita número de verdade (como na US-32).
  const valor = /^\d+([.,]\d{1,2})?$/.test(digitado) ? parseMoney(digitado) : NaN
  if (!UUID.test(donationId)) return { error: 'Doação não encontrada.' }
  if (!UUID.test(captadorId)) return { error: 'Escolha quem captou.' }
  if (!Number.isFinite(valor) || valor < 0) return { error: 'Informe a comissão em reais (por exemplo, 12,50).' }
  try {
    await setComissao(donationId, captadorId, Math.round(valor * 100) / 100)
  } catch (e) {
    console.error('salvarComissao error:', e)
    return { error: 'Não foi possível salvar a comissão.' }
  }
  revalidatePath('/relatorios/comissoes')
}

export async function removerComissao(donationId: string): Promise<{ error: string } | void> {
  if (!(await souCoordenacao())) return { error: AVISO_RESTRITO }
  if (!UUID.test(donationId)) return { error: 'Doação não encontrada.' }
  await removeComissao(donationId)
  revalidatePath('/relatorios/comissoes')
}
```

```tsx
'use client'

import { useMemo, useState, useTransition } from 'react'
import { alterarCaptador, cadastrarCaptador, removerComissao, salvarComissao } from '@/actions/comissoes'
import { formatBRL } from '@/lib/utils'
import type { Captador, DoacaoComComissao } from '@/types'

const dia = (iso: string) => new Date(iso).toLocaleDateString('pt-BR', { timeZone: 'America/Sao_Paulo', day: '2-digit', month: '2-digit' })

export function Comissoes({ doacoes, captadores }: { doacoes: DoacaoComComissao[]; captadores: Captador[] }) {

  const [erro, setErro] = useState<string | null>(null)
  const [marcadas, setMarcadas] = useState<Set<string>>(new Set())

    return [...m].sort((a, b) => a[0].localeCompare(b[0], 'pt-BR'))

    setErro(null)
    startTransition(async () => {
      const r = await acao()
      if (r?.error) setErro(r.error)
    })
  }

  return (
    <div className="flex flex-col gap-6 font-body">
      <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
        <div className="bg-paper border border-rule rounded-[16px] p-6">

        </div>
```
