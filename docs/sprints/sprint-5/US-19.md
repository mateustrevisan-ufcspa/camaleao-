# US-19 · Contribuição convertida em atendimentos (5 SP)

> Como apoiadora do Instituto, eu quero saber quantos atendimentos a minha contribuição ajudou a custear, para que eu entenda o efeito do que fiz, e não apenas o valor que paguei.

**Tipo:** user story · **Implementa:** Alissa · **Revisa:** Mateus · **Branch:** `us-19-atendimentos` · **Depende de:** nada · **Publica:** quinta 29/10, depois das 22h, depois do prazo de retorno das avaliadoras

## Critérios de aceitação

Do Product Backlog, com a parte do recibo indo para a US-16:

- [ ] Deve permitir configurar o valor de referência pela interface, iniciando em R$ 50
- [ ] Deve calcular atendimentos por venda, por doação e acumulado
- [ ] Deve exibir o número de atendimentos na conclusão da venda (no recibo, com a US-16)
- [ ] Deve permitir alterar o valor sem nova publicação do sistema
- [ ] Deve restringir a alteração ao perfil de coordenação e registrar quem alterou e quando

## O que muda para quem usa

- **Coordenação:** tela nova **Relatórios · Valor do atendimento** (`/relatorios/atendimento`), com o valor em vigor, um campo para alterar e a lista de quem alterou e quando. O topo de Relatórios ganha o link, com o valor atual.
- **Ficha da pessoa (coordenação):** o cartão "Contribuição no período" ganha um bloco "Em atendimentos custeados", com compras, doações e o total.
- **Balcão e coordenação:** depois de registrar uma venda, o painel do brechó mostra "Venda de R$ 30,00 registrada. Ela ajuda a custear 0,6 atendimento do Instituto."
- **Doações · Dinheiro (todos os perfis, cada um com as doações que já vê):** a lista ganha a coluna "Atendimentos", com o que cada doação ajuda a custear.

## Decisões

- **Cada alteração é uma linha nova** na tabela `valor_do_atendimento`. O valor em vigor é o da linha mais recente; as anteriores são o registro de quem alterou e quando. Ninguém altera nem apaga uma linha, nem a coordenação.
- **Quem e quando vêm do banco.** Um gatilho preenche `definido_por` com a conta logada e `definido_em` com a hora do banco, por cima do que o formulário mandar.
- **A conta usa sempre o valor em vigor,** também para contribuições antigas. Se a coordenação mudar de R$ 50 para R$ 60, a ficha passa a mostrar todo o histórico em R$ 60. É o mais simples de explicar ("quanto isso representa hoje"); se o Instituto preferir o valor da época, vira story nova.
- **Uma casa decimal,** com o plural do português: "0,6 atendimento", "1 atendimento", "1,5 atendimento", "2 atendimentos" (o plural começa no 2, e o zero vai no plural).
- **Por venda, por doação e acumulado:** por venda, no aviso depois de registrar; por doação, numa coluna da lista de doações em dinheiro, linha a linha; acumulado, na ficha (compras, doações em dinheiro e total no período). O recibo de cada doação (US-16) ainda não existe e vai usar as mesmas funções de `lib/atendimentos.ts`.
- **Por que coluna na lista, e não aviso depois da doação** (decidido em 26/10, na conferência do plano): o aviso pediria mudar o endereço depois da doação, que 6 testes de fumaça conferem (`/doacoes/dinheiro`). A coluna não muda endereço nem teste, e mostra o número de cada doação, também das antigas.
- **Sem item no menu:** a tela fica dentro de Relatórios, que já é só da coordenação no menu e no proxy (`/relatorios/atendimento` começa com `/relatorios`, que está em `SO_COORDENACAO`). Nada muda em `lib/rotas.ts` nem no `sidebar.tsx`.

## Parte A · Migração `supabase/migrations/20261028090000_valor_do_atendimento.sql`

```sql
-- US-19 · Contribuição convertida em atendimentos
-- Valor de referência de um atendimento do Instituto, informado pela coordenação
-- em 03/09/2026 (R$ 50). Cada alteração é uma linha nova: o valor em vigor é o da
-- linha mais recente, e as anteriores ficam como registro de quem alterou e quando.

-- ─────────────────────────────────────────
-- 1. Tabela. Ninguém altera nem apaga uma linha: alterar o valor é inserir outra.
-- ─────────────────────────────────────────

create table public.valor_do_atendimento (
  id           uuid primary key default gen_random_uuid(),
  valor        numeric(10,2) not null check (valor > 0 and valor <= 100000),
  definido_em  timestamptz not null default now(),
  definido_por uuid references public.users(id) on delete set null
);

create index idx_valor_do_atendimento_definido_em
  on public.valor_do_atendimento (definido_em desc);

-- ─────────────────────────────────────────
-- 2. Quem e quando vêm do banco, e não do formulário: o carimbo passa por cima
--    do que o cliente mandar. Na carga inicial (abaixo), auth.uid() é nulo.
-- ─────────────────────────────────────────

create or replace function public.carimbar_valor_do_atendimento()
returns trigger
language plpgsql
set search_path = ''
as $$
begin
  new.definido_em := now();
  new.definido_por := auth.uid();
  return new;
end;
$$;

create trigger carimbo_do_valor_do_atendimento
  before insert on public.valor_do_atendimento
  for each row execute function public.carimbar_valor_do_atendimento();

-- ─────────────────────────────────────────
-- 3. A equipe lê (o balcão precisa do valor para o aviso da venda); só a
--    coordenação grava. Sem política de update nem de delete: o histórico não muda.
-- ─────────────────────────────────────────

alter table public.valor_do_atendimento enable row level security;

create policy "equipe le" on public.valor_do_atendimento
  for select to authenticated using (public.eh_equipe());
create policy "coordenacao define" on public.valor_do_atendimento
  for insert to authenticated with check (public.eh_coordenacao());

revoke all on public.valor_do_atendimento from anon;

-- ─────────────────────────────────────────
-- 4. Valor inicial.
-- ─────────────────────────────────────────

insert into public.valor_do_atendimento (valor) values (50.00);
```

```ts
// US-19 · Contribuição convertida em atendimentos.
// O valor de um atendimento fica no banco (tabela valor_do_atendimento), e a
// coordenação altera pela tela Relatórios · Valor do atendimento.

// Valor informado pela coordenação em 03/09/2026. Só é usado se a tabela estiver vazia.
export const VALOR_INICIAL_DO_ATENDIMENTO = 50

// Quantos atendimentos um valor ajuda a custear, com uma casa decimal.
export function atendimentos(valor: number, valorDoAtendimento: number): number {
  if (!(valorDoAtendimento > 0) || !(valor > 0)) return 0
  return Math.round((valor / valorDoAtendimento) * 10) / 10
}

// "0,6 atendimento", "1 atendimento", "1,5 atendimento", "2 atendimentos".
// Em português o plural começa no 2, e o zero vai no plural.
export function textoDeAtendimentos(n: number): string {
  const numero = n.toLocaleString('pt-BR', { maximumFractionDigits: 1 })
  return `${numero} ${n === 0 || n >= 2 ? 'atendimentos' : 'atendimento'}`
}
```

```ts
// ── VALOR DO ATENDIMENTO (US-19) ──
// O valor em vigor é o da linha mais recente; as outras são o registro de quem alterou.
export async function getValorDoAtendimento(): Promise<number> {
  const supabase = await createClient()
  const { data, error } = await supabase
    .from('valor_do_atendimento')
    .select('valor')
    .order('definido_em', { ascending: false })
    .limit(1)
    .maybeSingle()
  if (error) throw new Error(error.message)
  return data ? Number(data.valor) : VALOR_INICIAL_DO_ATENDIMENTO
}

export async function getHistoricoDoValorDoAtendimento(): Promise<AlteracaoDoValorDoAtendimento[]> {
  const supabase = await createClient()
  const { data, error } = await supabase
    .from('valor_do_atendimento')
    .select('id, valor, definido_em, definido_por')
    .order('definido_em', { ascending: false })
    .limit(20)
  if (error) throw new Error(error.message)
  const linhas = data ?? []

  // Nomes de quem alterou, numa consulta só (a coordenação lê public.users).
  const ids = [...new Set(linhas.map((l) => l.definido_por).filter((id): id is string => Boolean(id)))]
  const nomes = new Map<string, string>()
  if (ids.length > 0) {
    const { data: usuarios, error: erroNomes } = await supabase.from('users').select('id, name').in('id', ids)
    if (erroNomes) throw new Error(erroNomes.message)
    for (const u of usuarios ?? []) nomes.set(u.id as string, u.name as string)
  }

  return linhas.map((l) => ({
    id: l.id as string,
    valor: Number(l.valor),
    definidoEm: l.definido_em as string,
    definidoPor: l.definido_por ? nomes.get(l.definido_por as string) ?? 'conta removida' : null,
  }))
}

export async function addValorDoAtendimento(valor: number): Promise<void> {
  const supabase = await createClient()
  // Quem e quando são preenchidos pelo banco (gatilho da migração da US-19).
  const { error } = await supabase.from('valor_do_atendimento').insert({ valor })
  if (error) throw new Error(error.message)
}
```

## Parte E · `actions/atendimento.ts` (arquivo novo)

```ts
'use server'

import { revalidatePath } from 'next/cache'
import { addValorDoAtendimento } from '@/lib/store'
import { parseMoney } from '@/lib/utils'

// US-19 · A coordenação altera o valor de um atendimento. O banco recusa quem não
// é coordenação (política "coordenacao define") e preenche quem e quando.
export async function definirValorDoAtendimento(fd: FormData): Promise<{ error: string } | void> {
  const digitado = ((fd.get('valor') as string) || '').trim()
  const valor = Math.round(parseMoney(digitado) * 100) / 100

  if (!digitado || !Number.isFinite(valor) || valor <= 0) return { error: 'Informe um valor maior que zero.' }
  if (valor > 100000) return { error: 'Confira o valor: ele passa de R$ 100.000,00.' }

  try {
    await addValorDoAtendimento(valor)
  } catch (e) {
    console.error('definirValorDoAtendimento error:', e)
    return { error: 'Não foi possível alterar o valor. Só a coordenação altera o valor do atendimento.' }
  }
  revalidatePath('/relatorios/atendimento')
  revalidatePath('/relatorios')
}
```

`tests/acesso/valor-do-atendimento.spec.ts`:

```ts
// US-19 · Valor do atendimento, conferido pela API.
import { test, expect } from '@playwright/test'
import type { SupabaseClient } from '@supabase/supabase-js'
import { CAIXA, COORDENACAO, entrarNaApi, recusado } from './apoio'

let coordenacao: SupabaseClient
let coordenacaoId: string
let caixa: SupabaseClient
let caixaId: string

test.describe.configure({ mode: 'serial' })

test.beforeAll(async () => {
  ;({ api: coordenacao, id: coordenacaoId } = await entrarNaApi(COORDENACAO))
  ;({ api: caixa, id: caixaId } = await entrarNaApi(CAIXA))
})

test.afterAll(async () => {
  // Ninguém apaga linha: o valor volta a R$ 50 com uma linha nova.
  if (coordenacao) await coordenacao.from('valor_do_atendimento').insert({ valor: 50 })
})

const vigente = (api: SupabaseClient) =>
  api.from('valor_do_atendimento').select('id, valor, definido_por, definido_em').order('definido_em', { ascending: false }).limit(1).single()

test('a caixa lê o valor, mas não grava', async () => {
  const leitura = await vigente(caixa)
  expect(leitura.error).toBeNull()
  expect(Number(leitura.data!.valor)).toBeGreaterThan(0)

  expect((await caixa.from('valor_do_atendimento').insert({ valor: 1 })).error).not.toBeNull()
})

test('a coordenação grava, e quem e quando vêm do banco', async () => {
  const r = await coordenacao
    .from('valor_do_atendimento')
    .insert({ valor: 55, definido_por: caixaId, definido_em: '2000-01-01T00:00:00Z' })
    .select('valor, definido_por, definido_em')
    .single()
  expect(r.error).toBeNull()
  expect(Number(r.data!.valor)).toBe(55)
  expect(r.data!.definido_por).toBe(coordenacaoId)
  expect(new Date(r.data!.definido_em).getFullYear()).toBeGreaterThan(2000)
})

test('ninguém altera nem apaga uma linha, nem a coordenação', async () => {
  const { data } = await vigente(coordenacao)
  const id = data!.id

  expect(recusado(await coordenacao.from('valor_do_atendimento').update({ valor: 1 }).eq('id', id).select())).toBe(true)
  expect(recusado(await coordenacao.from('valor_do_atendimento').delete().eq('id', id).select())).toBe(true)
  expect(recusado(await caixa.from('valor_do_atendimento').delete().eq('id', id).select())).toBe(true)

  const depois = await vigente(coordenacao)
  expect(Number(depois.data!.valor)).toBe(55)
})
```
