## 2. `lib/pagamento.ts` (arquivo novo)

O rótulo da forma existe hoje em três lugares (`lib/store.ts`, a página de relatórios e a rota de exportação), todos em código de servidor. A tabela do financeiro é componente do navegador e não pode importar `lib/store.ts`. Este arquivo não importa nada de servidor, então serve aos dois lados. Os outros três lugares ficam como estão nesta story.

```ts
// US-29 · Rótulo da forma de pagamento, para telas do navegador e do servidor.
// A venda guarda o nome interno (payment_methods.name).
const ROTULO_DA_FORMA: Record<string, string> = {
  pix: 'Pix',
  credit: 'Crédito',
  debit: 'Débito',
  cash: 'Dinheiro',
}

export function rotuloDaForma(forma: string): string {
  return ROTULO_DA_FORMA[forma] ?? forma
}

// "Crédito", "Crédito em 3x", "Débito", "Pix", "Dinheiro". Uma parcela é crédito à vista.
export function textoDoPagamento(venda: { payment_method: string; installments?: number | null }): string {
  const forma = rotuloDaForma(venda.payment_method)
  const parcelas = venda.installments ?? 1
  return venda.payment_method === 'credit' && parcelas > 1 ? `${forma} em ${parcelas}x` : forma
}
```

## 5. Teste: `tests/fumaca/forma-de-pagamento.spec.ts` (arquivo novo)

```ts
import { expect, test } from '@playwright/test'
import { entrar, valorUnico } from './apoio'

// US-29 · Forma de pagamento na conferência do recebimento.
test.describe('Fumaça: forma de pagamento no financeiro', () => {
  test('venda no crédito em 3x na SICREDI aparece com forma, parcelas e banco', async ({ page }) => {
    const valor = valorUnico(140)
    await entrar(page)

    await test.step('registrar a venda no crédito', async () => {
      await page.goto('/brecho/nova-venda')
      await page.getByPlaceholder('Categoria (saia, blusa, vestido...)').first().fill('vestido')
      await page.getByPlaceholder('0,00').first().fill(valor.digitado)
      await page.getByRole('button', { name: 'Crédito' }).click()
      await page.getByLabel('Banco / maquininha').selectOption('SICREDI')
      await page.getByLabel('Parcelas').selectOption('3')
      await page.getByRole('button', { name: 'Registrar venda' }).click()
      await expect(page).toHaveURL(/\/brecho\?venda=/)
    })

    await test.step('o financeiro mostra a forma, as parcelas e o banco', async () => {
      await page.goto('/brecho/financeiro')
      const pagamento = page.getByRole('row').filter({ hasText: valor.exibido }).getByTestId('pagamento-da-venda')
      await expect(pagamento).toContainText('Crédito em 3x')
      await expect(pagamento).toContainText('SICREDI')
    })
  })
})
```
