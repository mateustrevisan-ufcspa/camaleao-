# US-30 · Somas de vendas por dia e por semana (3 SP)

> Como coordenadora do financeiro, eu quero ver as somas de vendas por dia e por semana, além do mês, para que eu acompanhe o brechó no mesmo padrão da planilha que usamos hoje.

**Tipo:** user story · **Implementa:** Bibiana · **Revisa:** Laís · **Branch:** `us-30-somas-de-vendas` · **Depende de:** US-04a integrada (usa `dataDeBrasilia` de `lib/utils.ts`); dia de início da semana (pergunta à Nílian de 30/10; sem resposta, segunda-feira) · **Publica:** quinta 05/11, à noite

## Critérios de aceitação

- [ ] Para o mês escolhido, uma tabela com a soma de cada dia com venda e outra com a de cada semana, bruto e líquido, no horário de Brasília
- [ ] O total do mês é igual ao do relatório consolidado (US-17)
- [ ] A semana começa no mesmo dia da planilha que o Instituto usa hoje
- [ ] Só a coordenação vê; teste de acesso e teste de fumaça com venda às 23h30 de domingo (cai na semana certa)

## O que já existe

Retorno da Nílian (coordenação) em 29/10: a planilha de hoje tem somas por dia, por semana e por mês; o sistema só tem a do mês (relatório consolidado, US-17) e os cartões "Esta semana" e "Mês" da visão geral do brechó.

O relatório consolidado soma `sales_view.amount` com `sold_at` entre 0h do primeiro dia e 0h do dia seguinte ao último, no horário de Brasília (`brMidnightUtc` em `lib/store.ts`). Esta story usa o mesmo recorte, para o total do mês bater.

## Decisões

- **Tela nova, `/relatorios/vendas`,** com o mês escolhido (padrão: o mês corrente em Brasília). O prefixo `/relatorios` já está em `SO_COORDENACAO`: o proxy devolve o balcão, e o banco só devolve ao balcão as vendas de hoje de qualquer jeito.
- **Por dia:** só os dias com venda. **Por semana:** todas as semanas que tocam o mês, cortadas no primeiro e no último dia do mês, inclusive as sem venda (aparecem com zero). Assim a soma das semanas é o total do mês.
- **Início da semana numa constante** (`INICIO_DA_SEMANA`, 1 = segunda). Provisório até a Nílian responder.
- **Líquido:** o valor líquido gravado na venda. Venda sem líquido (a partir da US-32, quando a taxa não está cadastrada) entra com o bruto, como já faz o financeiro, e a tela avisa quantas são.
- **O cálculo fica numa lib sem banco** (`lib/vendas-por-periodo.ts`), no padrão de `lib/resumo-das-doacoes.ts` (BUG-02), para o teste conferir a virada do dia e da semana sem depender do relógio.

## Parte A · `lib/vendas-por-periodo.ts` (arquivo novo)

```ts
import { dataDeBrasilia } from './utils'

// US-30 · Somas de vendas por dia e por semana, no horário de Brasília.

// Dia em que a semana começa: 0 = domingo, 1 = segunda. Provisório: segunda-feira,
// até a Nílian dizer como é na planilha de hoje (pergunta de 30/10).
export const INICIO_DA_SEMANA = 1

export interface VendaParaSoma {
  sold_at: string
  amount: number
  // Sem taxa cadastrada (US-32), o líquido fica vazio e a soma usa o bruto.
  net_amount: number | null
}

export interface SomaDoPeriodo {
  inicio: string // AAAA-MM-DD, em Brasília
  fim: string
  vendas: number
  bruto: number
  liquido: number
  semLiquido: number
}

const centavos = (n: number) => Math.round(n * 100) / 100

function partes(dia: string): [number, number, number] {
  const [a, m, d] = dia.split('-').map(Number)
  return [a, m, d]
}

// Soma dias a uma data AAAA-MM-DD, sem fuso no meio.
export function somarDias(dia: string, n: number): string {
  const [a, m, d] = partes(dia)
  return new Date(Date.UTC(a, m - 1, d + n)).toISOString().slice(0, 10)
}

function diaDaSemana(dia: string): number {
  const [a, m, d] = partes(dia)
  return new Date(Date.UTC(a, m - 1, d)).getUTCDay()
}

export function inicioDaSemana(dia: string): string {
  return somarDias(dia, -((diaDaSemana(dia) - INICIO_DA_SEMANA + 7) % 7))
}

const vazia = (inicio: string, fim: string): SomaDoPeriodo => ({ inicio, fim, vendas: 0, bruto: 0, liquido: 0, semLiquido: 0 })

function somar(s: SomaDoPeriodo, v: VendaParaSoma) {
  s.vendas += 1
  s.bruto += Number(v.amount)
  s.liquido += Number(v.net_amount ?? v.amount)
  if (v.net_amount === null) s.semLiquido += 1
}

const arredondar = (s: SomaDoPeriodo): SomaDoPeriodo => ({ ...s, bruto: centavos(s.bruto), liquido: centavos(s.liquido) })

// Um item por dia com venda, em ordem.
export function somasPorDia(vendas: VendaParaSoma[]): SomaDoPeriodo[] {
  const porDia = new Map<string, SomaDoPeriodo>()
  for (const v of vendas) {
    const dia = dataDeBrasilia(v.sold_at)
    const s = porDia.get(dia) ?? vazia(dia, dia)
    somar(s, v)
    porDia.set(dia, s)
  }
  return [...porDia.values()].sort((a, b) => a.inicio.localeCompare(b.inicio)).map(arredondar)
}

// Todas as semanas que tocam o mês, cortadas no primeiro e no último dia do mês.
export function somasPorSemana(vendas: VendaParaSoma[], ano: number, mes: number): SomaDoPeriodo[] {
  const primeiro = `${ano}-${String(mes).padStart(2, '0')}-01`
  const ultimo = somarDias(`${mes === 12 ? ano + 1 : ano}-${String(mes === 12 ? 1 : mes + 1).padStart(2, '0')}-01`, -1)

  const semanas: SomaDoPeriodo[] = []
  for (let inicio = inicioDaSemana(primeiro); inicio <= ultimo; inicio = somarDias(inicio, 7)) {
    const fim = somarDias(inicio, 6)
    semanas.push(vazia(inicio < primeiro ? primeiro : inicio, fim > ultimo ? ultimo : fim))
  }

  for (const v of vendas) {
    const dia = dataDeBrasilia(v.sold_at)
    const semana = semanas.find((s) => s.inicio <= dia && dia <= s.fim)
    if (semana) somar(semana, v) // venda fora do mês não entra
  }
  return semanas.map(arredondar)
}

export function totalDe(somas: SomaDoPeriodo[]): { vendas: number; bruto: number; liquido: number; semLiquido: number } {
  return {
    vendas: somas.reduce((t, s) => t + s.vendas, 0),
    bruto: centavos(somas.reduce((t, s) => t + s.bruto, 0)),
    liquido: centavos(somas.reduce((t, s) => t + s.liquido, 0)),
    semLiquido: somas.reduce((t, s) => t + s.semLiquido, 0),
  }
}
```

// ── VENDAS POR DIA E POR SEMANA (US-30) ──
// O mesmo recorte do relatório consolidado: de 0h do dia 1 a 0h do mês seguinte, em Brasília.
export async function getVendasDoMes(ano: number, mes: number): Promise<VendaParaSoma[]> {
  const desde = brMidnightUtc(ano, mes, 1).toISOString()
  const ate = brMidnightUtc(ano, mes + 1, 1).toISOString() // Date.UTC passa de dezembro para janeiro
  const supabase = await createClient()
  const { data, error } = await supabase
    .from('sales_view')
    .select('sold_at, amount, net_amount')
    .gte('sold_at', desde)
    .lt('sold_at', ate)
    .order('sold_at')
  if (error) throw new Error(error.message)
  return (data ?? []).map((v) => ({
    sold_at: v.sold_at as string,
    amount: Number(v.amount),
    net_amount: v.net_amount === null ? null : Number(v.net_amount),
  }))
}

### `tests/fumaca/vendas-por-periodo.spec.ts` (arquivo novo)

```ts
import { expect, test } from '@playwright/test'
import { somasPorDia, somasPorSemana, totalDe } from '../../lib/vendas-por-periodo'
import { entrar, valorUnico } from './apoio'

// US-30 · Somas de vendas por dia e por semana.
const venda = (sold_at: string, amount: number, net_amount: number | null = amount) => ({ sold_at, amount, net_amount })

test('venda às 23h30 de domingo em Brasília fica no domingo e na semana do domingo', () => {
  // 01/11/2026 é domingo. 23h30 em Brasília é 02h30 de segunda em UTC.
  const vendas = [venda('2026-11-02T02:30:00.000Z', 50), venda('2026-11-02T15:00:00.000Z', 30)]

  const porDia = somasPorDia(vendas)
  expect(porDia.map((d) => d.inicio)).toEqual(['2026-11-01', '2026-11-02'])

  const semanas = somasPorSemana(vendas, 2026, 11)
  const doDomingo = semanas.find((s) => s.inicio <= '2026-11-01' && '2026-11-01' <= s.fim)!
  expect(doDomingo.bruto).toBeGreaterThanOrEqual(50)
  // Com a semana começando na segunda, o domingo 01/11 é uma semana só dele, cortada no início do mês.
  if (doDomingo.fim === '2026-11-01') expect(doDomingo.bruto).toBe(50)
})

test('as semanas cobrem o mês inteiro e somam o total do mês', () => {
  const vendas = [
    venda('2026-11-01T15:00:00.000Z', 10),
    venda('2026-11-15T15:00:00.000Z', 20, 19.5),
    venda('2026-11-30T23:00:00.000Z', 30, null), // sem taxa cadastrada: líquido pelo bruto
    venda('2026-12-01T03:30:00.000Z', 99), // 01/12 0h30 em Brasília: fora de novembro
  ]
  const semanas = somasPorSemana(vendas, 2026, 11)
  expect(semanas[0].inicio).toBe('2026-11-01')
  expect(semanas[semanas.length - 1].fim).toBe('2026-11-30')

  const total = totalDe(semanas)
  expect(total).toMatchObject({ vendas: 3, bruto: 60, liquido: 59.5, semLiquido: 1 })
})

test('a tela mostra a venda de hoje e o total igual ao do relatório consolidado', async ({ page }) => {
  const valor = valorUnico(60)
  await entrar(page)

  await page.goto('/brecho/nova-venda')
  await page.getByPlaceholder('Categoria (saia, blusa, vestido...)').first().fill('saia')
  await page.getByPlaceholder('0,00').first().fill(valor.digitado)
  await page.getByRole('button', { name: 'Registrar venda' }).click()
  await expect(page).toHaveURL(/\/brecho\?venda=/)

  await page.goto('/relatorios')
  const consolidado = (await page.getByTestId('relatorio-vendas').textContent())?.trim()

  await page.goto('/relatorios/vendas')
  await expect(page.getByTestId('total-do-mes')).toHaveText(consolidado!)
  await expect(page.getByTestId('vendas-por-dia').getByRole('row').last()).toBeVisible()
})
```
