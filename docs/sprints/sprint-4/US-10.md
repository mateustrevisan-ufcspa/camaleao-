# US-10 · Histórico unificado na ficha da pessoa (3 SP)

> Como coordenadora do Instituto, eu quero ver compras e doações da mesma pessoa em uma única tela, para que eu entenda a relação dela com o Instituto sem abrir três telas diferentes.

**Tipo:** user story · **Épico:** Ficha única (pedido N) · **Implementa:** Bibiana · **Revisa:** Mateus · **Branch:** `us-10-historico-unificado` · **Depende de:** nada. Não tem migração.

## Critérios de aceitação

- [ ] Deve listar compras e doações em ordem cronológica na ficha
- [ ] Deve identificar o tipo de cada contribuição
- [ ] Deve permitir abrir o registro original a partir do histórico

## O problema

A ficha da pessoa (tela Clientes) já mostra compras e doações, mas em dois cartões separados: "Histórico de compras" (as 5 mais recentes de no máximo 10) e "Histórico de doações" (as 6 mais recentes). Não dá para ver a sequência do que a pessoa fez, a lista corta o resto e nada leva ao registro completo.

## Decisões

- **Uma lista só,** com compras e os três tipos de doação (dinheiro, itens, tampinhas), da mais recente para a mais antiga. Mostra 10 de cada vez, com o botão "Mostrar mais".
- **Cada linha tem uma etiqueta colorida com o tipo:** Compra, Doação · dinheiro, Doação · itens, Doação · tampinhas.
- **"Abrir o registro original"** é um painel sobre a ficha com o registro como foi gravado na tabela de origem: a venda com as peças e o pagamento, ou a doação com todos os campos. Abre ao clicar na linha. O painel não mostra telefone: o telefone da pessoa já está na ficha.
- **A regra de acesso não muda.** Cada consulta passa pelas políticas de quem está logado: a coordenação vê o histórico completo; o balcão vê só o que foi lançado hoje (linha "Clientes: histórico" da matriz de permissões).
- **Tudo em arquivos da ficha.** A lista vira um componente novo (`historico-da-pessoa.tsx`). A US-08b mexe na tela Clientes, mas fora de `client-profile.tsx`, então as duas não se encostam.
- **A US-11 e a US-12 completam o painel depois.** Quando elas entrarem, o registro original passa a mostrar o peso da doação de item e o evento da doação em dinheiro. Os guias delas trazem esse passo.

## Como este guia mostra as mudanças

Arquivo novo vem inteiro. Em arquivo que já existe, o bloco mostra só o trecho: linha com `-` sai, linha com `+` entra, e as outras são contexto para achar o lugar.

```ts
// Compras e doações da pessoa numa lista só, da mais recente para a mais antiga.

const LIMITE_DO_HISTORICO = 500

const reais = (n: number) =>
  'R$ ' + Number(n).toLocaleString('pt-BR', { minimumFractionDigits: 2, maximumFractionDigits: 2 })

const dataEHora = (iso: string) =>
  new Date(iso).toLocaleString('pt-BR', { timeZone: 'America/Sao_Paulo', dateStyle: 'short', timeStyle: 'short' })

const texto = (v: unknown) => (v === null || v === undefined || v === '' ? 'não informado' : String(v))

const ROTULO_PAGAMENTO: Record<string, string> = { pix: 'PIX', credit: 'Crédito', debit: 'Débito', cash: 'Dinheiro' }

const UUID = /^[0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12}$/i

export async function getClientHistory(clientId: string): Promise<ItemDoHistorico[]> {
  const supabase = await createClient()
  const [vendas, dinheiro, itens, tampinhas] = await Promise.all([
    supabase.from('sales_view').select('id, sold_at, category, amount').eq('client_id', clientId).limit(LIMITE_DO_HISTORICO),
    supabase.from('donations_cash').select('id, donated_at, amount').eq('client_id', clientId).limit(LIMITE_DO_HISTORICO),
    supabase.from('donations_items').select('id, donated_at, category_name, quantity').eq('client_id', clientId).limit(LIMITE_DO_HISTORICO),
    supabase.from('donations_caps').select('id, donated_at, quantity, weight_kg').eq('client_id', clientId).limit(LIMITE_DO_HISTORICO),
  ])
  const erro = vendas.error ?? dinheiro.error ?? itens.error ?? tampinhas.error
  if (erro) throw new Error(erro.message)

  const linhas: ItemDoHistorico[] = [
    ...(vendas.data ?? []).map((v) => ({
      id: v.id,
      tipo: 'venda' as const,
      data: v.sold_at,
      descricao: v.category || 'Compra no brechó',
      valor: Number(v.amount),
    })),
    ...(dinheiro.data ?? []).map((d) => ({
      id: d.id,
      tipo: 'dinheiro' as const,
      data: d.donated_at,
      descricao: 'Doação em dinheiro',
      valor: Number(d.amount),
    })),
    ...(itens.data ?? []).map((d) => ({
      id: d.id,
      tipo: 'itens' as const,
      data: d.donated_at,
      descricao: d.quantity ? `${d.quantity}× ${d.category_name}` : d.category_name || 'Doação de itens',
      valor: null,
    })),
    ...(tampinhas.data ?? []).map((d) => ({
      id: d.id,
      tipo: 'tampinhas' as const,
      data: d.donated_at,
      descricao: d.weight_kg ? `${Number(d.weight_kg).toLocaleString('pt-BR')} kg de tampinhas` : `${d.quantity} tampinhas`,
      valor: null,
    })),
  ]
  return linhas.sort((a, b) => new Date(b.data).getTime() - new Date(a.data).getTime())
}

// O registro como foi gravado, para o painel "Registro original" da ficha.
// Devolve null quando o registro não existe ou o perfil de quem está logado não
// o enxerga (o balcão só lê o que foi lançado hoje).
export async function getRegistroOriginal(tipo: TipoDeContribuicao, id: string): Promise<RegistroOriginal | null> {
  if (!UUID.test(id)) return null
  const supabase = await createClient()

  if (tipo === 'venda') {
    const [venda, pecas] = await Promise.all([
      supabase.from('sales_view').select(COLUNAS_VENDA).eq('id', id).maybeSingle(),
      supabase.from('sale_items').select('category_name, amount').eq('sale_id', id).order('created_at'),
    ])
    if (venda.error) throw new Error(venda.error.message)
    if (pecas.error) throw new Error(pecas.error.message)
    const v = venda.data as Sale | null
    if (!v) return null
    const pagamento = [ROTULO_PAGAMENTO[v.payment_method] ?? v.payment_method, v.bank, v.installments ? `${v.installments}x` : '']
      .filter(Boolean)
      .join(' · ')
    return {
      tipo,
      titulo: 'Compra no brechó',
      campos: [
        { rotulo: 'Data', valor: dataEHora(v.sold_at) },
        { rotulo: 'Nome na venda', valor: texto(v.customer_name) },
        ...(pecas.data ?? []).map((p, i) => ({
          rotulo: `Peça ${i + 1}`,
          valor: `${texto(p.category_name)} · ${reais(p.amount)}`,
        })),
        { rotulo: 'Total', valor: reais(v.amount) },
        { rotulo: 'Pagamento', valor: texto(pagamento) },
        { rotulo: 'Valor líquido', valor: v.net_amount != null ? reais(v.net_amount) : 'não informado' },
        { rotulo: 'Recebimento conferido', valor: v.confirmed ? 'sim' : 'não' },
      ],
    }
  }

  if (tipo === 'dinheiro') {
    const { data, error } = await supabase.from('donations_cash_view').select(COLUNAS_DOACAO_DINHEIRO).eq('id', id).maybeSingle()
    if (error) throw new Error(error.message)
    const d = data as DonationCash | null
    if (!d) return null
    return {
      tipo,
      titulo: 'Doação em dinheiro',
      campos: [
        { rotulo: 'Data', valor: dataEHora(d.donated_at) },
        { rotulo: 'Nome na doação', valor: texto(d.donor_name) },
        { rotulo: 'Valor', valor: reais(d.amount) },
        { rotulo: 'Origem', valor: texto(d.origin) },
        { rotulo: 'Frequência', valor: d.frequency === 'monthly' ? 'mensal' : 'pontual' },
        { rotulo: 'Observações', valor: texto(d.notes) },
      ],
    }
  }

  if (tipo === 'itens') {
    const { data: d, error } = await supabase
      .from('donations_items')
      .select('donated_at, donor_name, category_name, quantity, condition, destination, notes')
      .eq('id', id)
      .maybeSingle()
    if (error) throw new Error(error.message)
    if (!d) return null
    return {
      tipo,
      titulo: 'Doação de itens',
      campos: [
        { rotulo: 'Data', valor: dataEHora(d.donated_at) },
        { rotulo: 'Nome na doação', valor: texto(d.donor_name) },
        { rotulo: 'Categoria', valor: texto(d.category_name) },
        { rotulo: 'Quantidade', valor: texto(d.quantity) },
        { rotulo: 'Estado', valor: d.condition === 'good' ? 'bom' : d.condition === 'needs_review' ? 'revisar' : 'não informado' },
        { rotulo: 'Destino', valor: d.destination === 'stock' ? 'estoque' : d.destination === 'direct' ? 'direto' : 'não informado' },
        { rotulo: 'Observações', valor: texto(d.notes) },
      ],
    }
  }

  if (tipo === 'tampinhas') {
    const { data: d, error } = await supabase
      .from('donations_caps')
      .select('donated_at, donor_name, quantity, weight_kg, notes')
      .eq('id', id)
      .maybeSingle()
    if (error) throw new Error(error.message)
    if (!d) return null
    return {
      tipo,
      titulo: 'Doação de tampinhas',
      campos: [
        { rotulo: 'Data', valor: dataEHora(d.donated_at) },
        { rotulo: 'Nome na doação', valor: texto(d.donor_name) },
        { rotulo: 'Quantidade', valor: texto(d.quantity) },
        { rotulo: 'Peso (kg)', valor: d.weight_kg ? Number(d.weight_kg).toLocaleString('pt-BR') : 'não informado' },
        { rotulo: 'Observações', valor: texto(d.notes) },
      ],
    }
  }

  // Tipo fora da lista (o valor chega do navegador): nada a mostrar.
  return null
}
```

- `getRegistroOriginal` recebe o tipo e o id que vêm do navegador. O id que não tem cara de UUID e o tipo fora da lista voltam `null`, sem consultar nada.

```tsx
export function HistoricoDaPessoa({ clientId }: { clientId: string }) {
  const [historico, setHistorico] = useState<{ linhas: ItemDoHistorico[]; erro: boolean } | null>(null)
  const [mostrando, setMostrando] = useState(POR_VEZ)
```

### C1. `tests/fumaca/historico-da-pessoa.spec.ts` (arquivo novo)

```ts
import { expect, test } from '@playwright/test'
import { entrar } from './apoio'

// US-10 · Histórico unificado na ficha. Usa a Roberta Lima do seed:
// duas compras e uma doação em dinheiro de R$ 100,00.
test.describe('Fumaça: histórico unificado na ficha', () => {
  test('compras e doações numa lista só, com o tipo e o registro original', async ({ page }) => {
    await entrar(page)
    await page.goto('/clientes')
    await page.getByPlaceholder('Buscar cliente...').fill('Roberta')
    await page.getByText('Roberta Lima').first().click()
    await expect(page.getByRole('heading', { name: 'Roberta Lima' })).toBeVisible()

    const historico = page.getByTestId('historico-da-pessoa')

    await test.step('a lista tem compra e doação, cada uma com o tipo', async () => {
      await expect(historico.getByText('Compra', { exact: true }).first()).toBeVisible()
      await expect(historico.getByText('Doação · dinheiro', { exact: true }).first()).toBeVisible()
    })

    await test.step('a lista vem da mais recente para a mais antiga', async () => {
      const datas = await historico.getByTestId('data-do-registro').allTextContents()
      const emMs = datas.map((d) => {
        const [dia, mes, ano] = d.trim().split('/').map(Number)
        return Date.UTC(ano, mes - 1, dia)
      })
      expect(emMs.length).toBeGreaterThanOrEqual(3)
      expect(emMs).toEqual([...emMs].sort((a, b) => b - a))
    })

    await test.step('o registro original abre a partir da lista', async () => {
      await historico.getByText('Doação · dinheiro', { exact: true }).first().click()
      const painel = page.getByRole('dialog', { name: 'Registro original' })
      await expect(painel.getByRole('heading', { name: 'Doação em dinheiro' })).toBeVisible()
      await expect(painel.getByText('R$ 100,00')).toBeVisible()
      await painel.getByRole('button', { name: 'Fechar' }).click()
      await expect(painel).toHaveCount(0)
    })
  })
})
```
