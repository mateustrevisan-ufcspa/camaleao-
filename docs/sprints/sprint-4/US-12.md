# US-12 · Doação vinculada a um evento (5 SP)

> Como coordenadora do Instituto, eu quero registrar como doação o valor pago em um evento, como ingresso de festa ou jantar, para que a contribuição feita em evento entre na conta do que a pessoa apoiou no ano.

**Tipo:** user story · **Épico:** Doações reorganizadas · **Implementa:** Laís · **Revisa:** Alissa · **Branch:** `us-12-doacao-de-evento` · **Depende de:** nada no banco. Mexe nos mesmos arquivos de doação que a US-11 (também da Laís): comece esta depois de abrir o PR da US-11 e traga a `main` assim que a US-11 for integrada.

## Critérios de aceitação

- [ ] Deve permitir cadastrar um evento com nome e data
- [ ] Deve registrar doação em dinheiro vinculada a um evento, com o valor pago
- [ ] Deve mostrar o total arrecadado por evento
- [ ] Deve manter a doação em dinheiro sem evento funcionando como hoje

## O problema

O Instituto faz festas, jantares e outros eventos com ingresso, e pediu em 03/09 que esse valor entre como doação. Hoje não existe evento no sistema: o ingresso ou vira uma doação em dinheiro sem indicação de onde veio, ou não é registrado.

## Decisões

- **Tabela nova `events`** com nome, data e situação (ativo ou não). A equipe lê, para escolher o evento na doação; só a coordenação cadastra, altera e apaga. Evento com doação não pode ser apagado (chave estrangeira com `restrict`): para tirar da lista, desativa.
- **A doação em dinheiro ganha `event_id`, opcional.** Sem evento, ela continua exatamente como hoje (quarto critério). O formulário ganha um campo "Evento (opcional)" com os eventos ativos.
- **O total por evento vem da view `events_totals`,** com `security_invoker`: respeita as políticas de quem consulta. A tela de eventos é só da coordenação, que vê todas as doações; para o balcão, a view somaria só as doações de hoje.
- **A `donations_cash_view` é recriada** para trazer o nome do evento, mantendo `security_invoker` e as permissões. Nenhuma outra story desta sprint mexe nela.
- **Tela em `/doacoes/eventos`** (já em `lib/rotas.ts`): cadastro do evento e lista com data, número de doações e total arrecadado. O menu de Doações ganha "Eventos", só para a coordenação.
- **A lista de doações em dinheiro mostra o evento** de cada doação, e a exportação ganha a coluna.
- **Doação de campanha (US-13) não entra:** é a próxima no roadmap e a terceira na ordem de corte. A tabela `events` não tenta servir às duas.

```sql
-- US-12 · Doação vinculada a um evento
-- Eventos do Instituto (festa, jantar) e doação em dinheiro ligada a um deles.

-- ─────────────────────────────────────────
-- 1. Eventos. A equipe lê (para escolher na doação); a coordenação cadastra,
--    altera e apaga. Evento com doação não é apagado: desativa.
-- ─────────────────────────────────────────

create table public.events (
  id         uuid primary key default gen_random_uuid(),
  name       text not null check (btrim(name) <> ''),
  event_date date not null,
  active     boolean not null default true,
  created_at timestamptz not null default now(),
  created_by uuid default auth.uid() references public.users(id) on delete set null
);

alter table public.events enable row level security;

create policy "equipe le" on public.events
  for select to authenticated using (public.eh_equipe());
create policy "coordenacao cadastra" on public.events
  for insert to authenticated with check (public.eh_coordenacao());
create policy "coordenacao altera" on public.events
  for update to authenticated using (public.eh_coordenacao()) with check (public.eh_coordenacao());
create policy "coordenacao apaga" on public.events
  for delete to authenticated using (public.eh_coordenacao());

revoke all on public.events from anon;

-- ─────────────────────────────────────────
-- 2. Doação em dinheiro com evento, opcional. Sem evento, tudo como antes.
-- ─────────────────────────────────────────

alter table public.donations_cash
  add column if not exists event_id uuid references public.events(id) on delete restrict;

create index if not exists idx_dcash_event_id on public.donations_cash (event_id);

-- ─────────────────────────────────────────
-- 3. A view de doações em dinheiro passa a trazer o evento. Como "create or
--    replace view" não deixa mudar colunas no meio, ela é recriada, com
--    security_invoker (sem ele a view ignora as políticas: US-23).
-- ─────────────────────────────────────────

drop view if exists public.donations_cash_view;
create view public.donations_cash_view
with (security_invoker = true) as
select
  d.id,
  d.client_id,
  d.donor_name,
  d.donor_phone,
  d.amount,
  d.frequency,
  d.donated_at,
  d.notes,
  d.created_at,
  coalesce(o.name, '') as origin,
  d.event_id,
  coalesce(e.name, '') as event_name
from public.donations_cash d
left join public.cash_origins o on o.id = d.origin_id
left join public.events e on e.id = d.event_id;

-- ─────────────────────────────────────────
-- 4. Total arrecadado por evento. security_invoker: soma só as doações que
--    quem consulta enxerga (a coordenação, todas).
-- ─────────────────────────────────────────

create view public.events_totals
with (security_invoker = true) as
select
  e.id,
  e.name,
  e.event_date,
  e.active,
  count(d.id)::int as donation_count,
  coalesce(sum(d.amount), 0) as total
from public.events e
left join public.donations_cash d on d.event_id = e.id
group by e.id;

-- Uma view recriada perde as permissões da anterior. Só quem está logado lê.
revoke all on public.donations_cash_view, public.events_totals from anon;
grant select on public.donations_cash_view, public.events_totals to authenticated;
```

```ts
// ── EVENTOS (US-12) ──
// A equipe lê os eventos para escolher na doação; só a coordenação cadastra e altera.

export async function getEventosAtivos(): Promise<Evento[]> {
  const supabase = await createClient()
  const { data, error } = await supabase
    .from('events')
    .select('id, name, event_date, active')
    .eq('active', true)
    .order('event_date', { ascending: false })
  if (error) throw new Error(error.message)
  return (data ?? []) as Evento[]
}

export async function getEventosComTotal(): Promise<EventoComTotal[]> {
  const supabase = await createClient()
  const { data, error } = await supabase
    .from('events_totals')
    .select('id, name, event_date, active, donation_count, total')
    .order('event_date', { ascending: false })
  if (error) throw new Error(error.message)
  return ((data ?? []) as EventoComTotal[]).map((e) => ({ ...e, total: Number(e.total) }))
}

export async function addEvento(name: string, eventDate: string): Promise<void> {
  const supabase = await createClient()
  const { error } = await supabase.from('events').insert({ name, event_date: eventDate })
  if (error) throw new Error(error.message)
}

export async function setEventoAtivo(id: string, active: boolean): Promise<void> {
  const supabase = await createClient()
  // Com a política barrando, o update não dá erro: só não altera nada.
  const { data, error } = await supabase.from('events').update({ active }).eq('id', id).select('id')
  if (error) throw new Error(error.message)
  if (!data || data.length === 0) throw new Error('Evento não encontrado ou sem permissão para alterar.')
}
```

```ts
export async function criarEvento(fd: FormData): Promise<{ error: string } | void> {
  const name = ((fd.get('name') as string) || '').trim()
  const date = ((fd.get('event_date') as string) || '').trim()
  if (!name) return { error: 'Informe o nome do evento.' }
  if (name.length > 80) return { error: 'Use um nome de até 80 letras.' }
  if (!DATA.test(date)) return { error: 'Informe a data do evento.' }

    await addEvento(name, date)
  } catch (e) {
    console.error('criarEvento error:', e)
    return { error: 'Não foi possível cadastrar o evento. Tente novamente.' }
  }
  revalidarTelas()
}

export async function alternarEvento(id: string, ativo: boolean): Promise<{ error: string } | void> {
```
